- the token is at `Authroization` header and perfixed with `Bearer` --> `Authorization: bearer TOKEN` 
```java
@Service  
public class JwtService {  
  
    final static String secret = "secretasfasdfasdfasdfasdfasdfasdfasdfasdfasdfasdfasdfasdfasfd";  
    final static long tokenExpiration = 86400;  
      
	public boolean validateToken(String authHeader){  
	    var token = authHeader.replaceFirst("Bearer ", "");  
	    try {  
	        var claims = Jwts.parser()  
	                .verifyWith(Keys.hmacShaKeyFor(secret.getBytes()))  
	                .build()  
	                .parseSignedClaims(token)  
	                .getPayload(); // if we have invalid token, this will throw exception.  
	        return claims.getExpiration().after(new Date()); // check if token is still in valid time.  
	    }catch (JwtException exception){  
	        return false;  
	    }  
	}  
}
```
