@Service  
public class JwtService {  
  
    final static long tokenExpiration = 86400;  
  
    public String generateToken(String email){  
        return Jwts.builder()  
                .subject(email) // subject is something unique for each object like ID.  
                .issuedAt(new Date())  
                .expiration(new Date(System.currentTimeMillis() + 1000*tokenExpiration))  
                .signWith(Keys.hmacShaKeyFor("secret".getBytes()))  
                .compact(); // this build th object == build();  
    }  
}
